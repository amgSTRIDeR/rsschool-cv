Cleaning up multiple event listeners with a single AbortController. 

const controller = new AbortController();
const { signal } = controller;

window.addEventListener(
  'resize',
  handleResize,
  { signal }
);

document.addEventListener(
  'keydown',
  handleKeydown,
  { signal }
);

button.addEventListener(
  'click',
  handleClick,
  { signal }
);

// Removes all listeners associated with this signal
